% Lệnh tự động sắp xếp và in danh mục ra tệp PDF (Đã sửa lỗi lệch tâm)
\usepackage{varwidth}
\newcommand{\printabbreviations}{%
  \DTLsort{key}{abbreviations}% Sắp xếp các từ viết tắt theo thứ tự bảng chữ cái ABC
  \begin{center}
    \renewcommand{\arraystretch}{1.3} % Tạo khoảng cách thoáng giữa các dòng bảng
    \begin{tabular}{@{} l @{\hspace{1.5cm}} l @{}} % Trả về cột l để hệ thống tự tính kích thước thực tế
      \DTLforeach{abbreviations}{\thekey=key,\thedescription=description}{%
        \thekey & \begin{varwidth}{10cm}\thedescription\end{varwidth} \\ % Giới hạn tối đa 10cm, ngắn hơn tự co lại
      }%
    \end{tabular}
  \end{center}
}
